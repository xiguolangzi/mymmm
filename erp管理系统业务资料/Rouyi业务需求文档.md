```bash
# 1 商品
	-- 主要维护sku公共数据：商品分类、商品名称、品牌、计量单位、税率、成本计算方式
	-- 主图: 后期网站使用

	-- SKU为最小单位，业务使用SKU不适用"商品"，"商品"主要是服务 SKU 而准备的；
	-- SKU规格：
		-- SKU编码：唯一独立的编码，用于识别产品；
		-- 规格名称： 单独维护(颜色、尺寸...)
		-- 规格属性值：红 黑 S M L ...
		-- 每个规格有独立的图片
		-- 每个规格有独立的价格：SKU单价、单价1、单价2、单价3、单价4、单价5、单价6
		-- 每个SKU有自己的 启售/停售 状态，
			-- 如果"商品"停售，其他所有SKU停售
			-- 如果"商品"启售，其他所有SKU启售
		-- 显示每个SKU的库存；

# 3 商品(没有SKU)
	-- 会自动生成一个默认的SKU；
	-- 默认的SKU的 skuValue = "default";
	-- 默认的SKU单价、状态同步 "商品"；
	-- SKU规格显示："--"
	
# 4 处理skuValue的展示：
	不需要JSON解析，直接通过方法 getSkuValue() 获取，自动处理默认和非默认等情况，如下：
	<div v-if="getSkuValue(scope.row.skuValue) === 'default'">
     	--  <!-- 直接显示默认 SKU -->
     </div>
     <div v-else v-for="(item, index) in getSkuValue(scope.row.skuValue)" :key="index">
      	<strong>{{ item[0] }}:</strong>
      	<span>{{ item[1] }}</span>
      </div>
    
# 5 基础单价：
	-- SKU单价 是基础单价！
	-- 单价2、单价3、单价4、单价5、单价6 都是备选单价使用！
```

```bash
	-- 3.4 过期日期：手动选择！
	-- 3.5 展示 绑定采购订单的订单总数量、订单欠货数量；
```

#### 	6.2.1 业务数据

```bash
# 1 交班记录信息
	-- 交班ID
	-- 交班编号
	-- 交班cajaID
	-- 收银员Id
	-- 上次交班Id：交班的现金余额
	-- 交班时间：开始/结束时间
	-- 交班的总收入(销售单金额 - 退货单金额)
	-- 现金收入(正数)
	-- 银行收入
	-- 现金退货(负数)
	-- 银行退货
	-- 找零(负数)
	-- 优惠抹零(负数)
	-- 应交金额 = 现金收入 + 现金退货 - 找零 - 优惠抹零
	-- 补充金额(正数)
	-- 上交金额(负数)
	-- 长/短款金额 
		-- 正数 长款(多出的现金)
		-- 负数 短款(缺失的现金)
	-- 本次交班余额 = 应交金额 + 补充金额 - 上交金额 +/- 长/短款金额
	-- 盘点现金总金额 = 盘点现金数据的合计
		-- 交班必须满足条件：本次交班余额 == 盘点现金总金额
	
# 2 交班财务结转：
	-- 交班总收入 ->
		借：库存现金-交班 100
			银行存款-交班 200
			销售费用-收银抹零 20
			贷：主营业务收入 320
	-- 补充金额：
		借：库存现金-交班 30
			贷：库存现金 30
	-- 上交金额：
		借：库存现金 125
			贷：库存现金-交班 125
	-- 短款：
		借：管理费用-交班短款 5
			贷：库存现金-交班 5
	-- 长款：
		借：库存现金-交班 10
			贷：营业外收入-交班长款 10
			
	-- 现金银行日记账：
		-- 库存现金/银行存款 <-> 库存现金-交班/银行存款-交班
		-- 正常 "银行存款-交班" 是没有余额的
		-- 如果存在余额就要核对银行到款记录
		************  每日必须核对信息 start ****************
		-- 交班时的银行收款记录表
		-- 交班时银行收款机小票记录
		-- 注意：以上两个步骤，避免收银人员 将现金收银操作成银行收银，交班后现金部分私自带走！
		-- 银行到账信息 -> 记录现金银行日记账 / 财务费用-caja刷卡未到账(备注小票流水、核对原因)
		************  每日必须核对信息 end ****************
```

#### 	6.2.2 业务流程

```bash
# 1 交班的业务流程 
	-> 1 没有注册无法访问！
	-> 2 注册后获取收银台信息(cajaId cajaName) -> 获取上一次成功交班的信息：
		-> 2.1 没有上一次交班信息
			-> 2.1.1 开始值班；
		-> 2.2 有未完成的交班信息 -> 校验是不是同一人值班
        	-> 2.2.1 是同一人
        		-> 继续值班；
        		-> 交班；
        	-> 2.2.2 不是同一人(租户管理员除外)
        		-> 提示、并退出收银界面；
		-> 2.3 有已完成的交班信息
			-> 2.3.1 开始值班 -> 开始值班之前需要 盘点总金额 = 上次值班的现金余额 ？ 允许开始值班 ： 不允许开始值班 ；
# 2 开始值班流程：
	-> 2.1 盘点现金余额，核对盘点总金额 = 上一次交班的余额 才允许开始值班；
# 3 交班流程：
	-> 3.1 盘点现金余额，核对盘点金额 = 本次交班余额 才允许交班；
	-> 3.2 本次交班余额 = 应交现金 + 补充现金 - 上交现金 + 长短款(正为长款，负为短款)；
	-> 3.3 可编辑金额：
		-> 3.3.1 补充现金: 由财务给收银台补充的现金(现金银行日记账)；
		-> 3.3.2 上交现金: 上交给财务的现金(现金银行日记账)；
		-> 3.3.3 长短款：用来调整本次交班余额(销售费用/其他营业收入)；
		-> 3.3.4 盘点金额：方便并核对交班现金余额；
# 4 发现上一次值班的现金余额与盘点金额不一致的问题处理方式：
	-> 4.1 出错人员直接自掏腰包补齐现金，线下不需要其他操作，只要金额对的上就好；
	-> 5.2 管理人员修改上一次的值班信息，修改长短款金额，写好备注信息；
# 5 长短款作用：
	-> 5.1 记录收银工作人员失误的金额，可以月统计；
	-> 5.2 当值班人员每月失误超过一定金额，可以给与一些警告措施；
	-> 5.3 最后长短款将计入"销售费用-交班费用"中；
# 6 交班的数据统计：
	-- 计算销售/退货订单：
		-- 订单类型 != 已拆单
			-- 预购 、 确认 、已生成发票
    	-- 订单状态 != 已取消的
    		-- 预购、确认
    	-- shiftId 、tenantId
	-- 现金收入(正数)：
		-- 订单方向 = 1 销售订单
	-- 银行收入(正数)：
		-- 订单方向 = 2 销售订单
	-- 现金退货(负数)：
		-- 订单方向 = 2 退货订单
	-- 银行退货(负数)：
		-- 订单方向 = 2 退货订单
	-- 找零(负数)：
        -- 订单方向 = 1 销售订单
        -- 订单方向 = 2 退货订单 ，注意：退货订单不开启"找零"功能
	-- 优惠抹零(负数)：
		-- 订单方向 = 1 销售订单
        -- 订单方向 = 2 退货订单 ，注意：退货订单不开启"优惠抹零"功能

```

#### 			6.6.2 二批模式

```bash
# 0 订单类型 order_type：
	-- 1 预购订单：只有初始单号 order_init_no 没有订单号的订单 order_no
	-- 2 确认订单：生成订单号 order_no
	-- 3 已拆单：生成新的订单，新订单绑定原始订单，原始订单的类型修改成 "已拆单"
	-- 4 生成发票：生成发票数据的订单
	统计销售数量：预购订单 + 确认订单 + 生成发票
	应收统计：预购订单金额 + 确认订单金额 + 发票金额(生成发票)
# 1 订单状态 order_status：
	--	1-预购，2-拣货，3-复核，4-确认订单，5-已取消
	
# 2 支付状态 order_pay_status：
	--	1-现结(当场结清) 2-欠款(未结清、部分结清) 
	--	手动开单 默认 欠款
	--	caja 默认 现结
	
# 3 挂单 order_is_hold
	--	预购、拣货、复核 阶段都可以挂起；
	--	挂单状态 order_is_hold：0-正常订单 1-挂单；
	--	拣货、复核 阶段如果有异常，支持文员修改预购订单；
```

#### 	7.6.1 业务设计

```bash
# 1 维护库位
	-- 根据仓库维护库位层级关系；
	-- 最多5级，每级的库位编码规则遵顼：xx-xx-xx-xx-xx；
	-- 库存编码只能是数字或字母；
# 2 库位参数：
	-- 长、宽、高 = 体积；
	-- 最大承受重量；
	-- 状态：空闲、占用、锁定、禁用
	-- 类型：
		-- 拣货库位：订单消除数量的库位
		-- 灵活库位：高架货位，非拣货库位，上下架可以变更绑定的SKU；
		-- 固定库位：高架货位，非拣货库位，上下架无法变更绑定的SKU，只能后台变更绑定关系；
     -- 绑定SKU：
     	-- skuID
     	-- 批次号
     	-- 过期时间
     	-- 存储数量
     	-- 最大存储数量
     -- 报警参数：
     	-- 数量下限；
     	-- 过期时间下限；
     -- 排序权重：
     	-- 销售订单 按照排序权重进行排序；
     	-- 帮助拣货人员规划路线；
     	-- 根据库位编号自动生成，简易手动维护；
     -- 库位绑定sku:
     	-- "唯一标识"：sku+批次号
     	-- 同一库位只能绑定一个"唯一标识"
# 3 使用场景：
	-- 首先客户需要启用库位；
	-- 启用库位，订单扣除扣除库位库存
	
# 4 启用库位(二批)：
	-- 销售订单 处理：
		-- 销售订单中的sku 生成库位信息(库位编码、库位权重、批次、过期时间)；
		-- "拣货状态"的订单直接记录"预扣除"库位数量，根据与扣除后的结果 -> 预警补货信息；
		-- 如果同一个sku存放多个位置，根据 过期时间、批次号 进行先进先出处理！
		-- 如果一个sku存放多个位置，会出现 同一个sku去多个地方取货的情况
	-- 库位处理：
		-- 数量分为：现存量、预扣除数量、可使用数量；
		-- 通过"可使用数量"进行预警、排单！
	-- 库位安排：
		-- 1层为拣货库位；
		-- 2层对应一层为固定库位(存放与1层相同SKU的托盘，方便捕获)；
		-- 3层以上为活动库位；
		-- 库位编码排序，与拣货顺序一致，蛇形排序！
	-- 库位数量扣除：
		-- 订单先计入冻结库位 -> 拣货扣除冻结数量 + 现存数量 -> "可使用数量" = "现存数量" - "冻结数量"
	-- 库位上架：
		-- 入库后 仓库"库存数量" 为历史原始依据；
        -- 字段"+未上架数量"="+入库数量"
        -- "库存数量" = "未上架数量" + "库位数量"
    -- 库位补货：
    	-- 已上架的库位之间进行数量调整；
    	-- 扫码库位、扫码 (sku+批次) -> 出库数量
    	-- 扫码库位、扫码 (sku+批次) -> 入库数量
    	-- "补货单"：最后的出入库数量必须 = 0
    -- 订单扣除库存：
    	-- 订单只能从拣货库位扣除；
    	-- "拣货库位"扣除 -> 会同步扣除"库存源数据"的"库存"
# 5 关闭库位(二批)
	-- "库存源数据" 的 "未上架数量" 清空；
	-- "入库数量" 不再同步 "未上架数量";
	-- 订单不在生成库位信息；
	-- 订单直接扣除 "库存源数据" 的 "库存数量"；
	
# 6 使用过程中开启库位：
	-- "库存源数据" 的 "未上架数量" 同步 "库存数量"；
	-- "入库数量" 同步 "未上架数量";
	-- 盘点库位："唯一标识"维度 -> 系统自动维护 "未上架数量" = "库存数量" - "库位现存量"
# 7 客户设置：
	-- 关闭库位管理/开启库位管理；
	-- 开启/关闭 批次：(关闭批次，批次为null)
	
```
